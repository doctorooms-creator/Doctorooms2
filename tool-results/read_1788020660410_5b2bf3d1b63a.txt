   445→// ─── Search Section Sub-component ────────────────────────────────────
   446→// Doctor can search by Appointment ID (shows that 1 visit's prescription)
   447→// or by Mobile Number (shows ALL prescriptions for that patient across all visits).
   448→
   449→function SearchSection() {
   450→  const [searchType, setSearchType] = useState<'appointment' | 'mobile'>('appointment')
   451→  const [searchValue, setSearchValue] = useState('')
   452→  const [searchResults, setSearchResults] = useState<{
   453→    patientName: string
   454→    patientMobile: string
   455→    prescriptions: {
   456→      id: string
   457→      appointmentNo: string
   458→      disease: string
   459→      date: string
   460→      status: string
   461→      medicinesCount: number
   462→      description: string
   463→      bookingId: string
   464→    }[]
   465→  } | null>(null)
   466→  const [searching, setSearching] = useState(false)
   467→  const [searched, setSearched] = useState(false)
   468→
   469→  const handleSearch = async () => {
   470→    if (!searchValue.trim()) {
   471→      toast.error('Please enter a value to search')
   472→      return
   473→    }
   474→    setSearching(true)
   475→    setSearched(true)
   476→    try {
   477→      const res = await fetch(
   478→        `/api/dashboard/doctor/search-prescriptions?type=${searchType}&value=${encodeURIComponent(searchValue.trim())}`
   479→      )
   480→      if (!res.ok) {
   481→        const data = await res.json().catch(() => ({}))
   482→        throw new Error(data.error || 'Search failed')
   483→      }
   484→      const data = await res.json()
   485→      setSearchResults(data)
   486→    } catch (err: unknown) {
   487→      toast.error(err instanceof Error ? err.message : 'Search failed')
   488→      setSearchResults(null)
   489→    } finally {
   490→      setSearching(false)
   491→    }
   492→  }
   493→
   494→  return (
   495→    <Card className="border-teal-200/50 dark:border-teal-900/30">
   496→      <CardHeader className="pb-3">
   497→        <CardTitle className="text-base font-semibold flex items-center gap-2">
   498→          <FileSearch className="h-4 w-4 text-teal-600 dark:text-teal-400" />
   499→          Search Patient History
   500→        </CardTitle>
   501→      </CardHeader>
   502→      <CardContent className="space-y-4">
   503→        {/* Search type toggle */}
   504→        <div className="flex gap-2">
   505→          <Button
   506→            variant={searchType === 'appointment' ? 'default' : 'outline'}
   507→            size="sm"
   508→            onClick={() => { setSearchType('appointment'); setSearchResults(null); setSearched(false) }}
   509→            className={searchType === 'appointment' ? 'bg-teal-600 hover:bg-teal-700' : ''}
   510→          >
   511→            By Appointment ID
   512→          </Button>
   513→          <Button
   514→            variant={searchType === 'mobile' ? 'default' : 'outline'}
   515→            size="sm"
   516→            onClick={() => { setSearchType('mobile'); setSearchResults(null); setSearched(false) }}
   517→            className={searchType === 'mobile' ? 'bg-teal-600 hover:bg-teal-700' : ''}
   518→          >
   519→            By Mobile Number
   520→          </Button>
   521→        </div>
   522→
   523→        {/* Search input */}
   524→        <div className="flex gap-2">
   525→          <Input
   526→            placeholder={searchType === 'appointment' ? 'Enter Appointment ID (e.g. APMT-496)' : 'Enter Mobile Number (e.g. 9876543210)'}
   527→            value={searchValue}
   528→            onChange={(e) => setSearchValue(e.target.value)}
