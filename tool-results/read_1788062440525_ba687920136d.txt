    98→  const queryClient = useQueryClient()
    99→  const doctorId = params.doctorId as string
   100→

   103→  const departmentId = searchParams.get('departmentId') || ''
   104→
   105→  // State
   106→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(undefined)
   107→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
   108→  const [bookingMode, setBookingMode] = useState<'InPerson' | 'VideoCall'>('InPerson')
   109→  const [disease, setDisease] = useState('')
   110→  const [description, setDescription] = useState('')
   111→  const [bookingState, setBookingState] = useState('')
   112→  const [bookingCity, setBookingCity] = useState('')
   113→  const [bookingGender, setBookingGender] = useState('')
   114→  const [showBookingForm, setShowBookingForm] = useState(false)
   115→  const { user } = useAuthStore()
   116→  const [slotStatuses, setSlotStatuses] = useState<SlotAvailability[]>([])
   117→
   118→  // ── Fetch doctor info ──
   119→  const {
   120→    data: doctorData,
   121→    isLoading: doctorLoading,
   122→  } = useQuery({
   123→    queryKey: ['doctor-info', doctorId],
   124→    queryFn: () => fetch(`/api/doctors/${doctorId}`).then((r) => {
   125→      if (!r.ok) throw new Error('Failed to load doctor info')
   126→      return r.json()
   127→    }),
   128→    enabled: !!doctorId,
   129→    retry: 1,
   130→  })
   131→
   132→  const doctor: DoctorInfo | null = doctorData?.doctor || null
   133→
   134→  // ── Fetch doctor schedule (public, no auth) ──
   135→  const {
   136→    data: scheduleData,
   137→    isLoading: scheduleLoading,
   138→  } = useQuery<ScheduleData>({
   139→    queryKey: ['doctor-schedule', doctorId],
   140→    queryFn: () => fetch(`/api/doctors/${doctorId}/schedule`).then((r) => r.json()),
   141→    enabled: !!doctorId,
   142→  })
   143→
   144→  const fees = doctor?.doctor?.fees ?? scheduleData?.fees ?? 0
   145→
   146→  // ── Determine available days for the selected date ──
   147→  const scheduleForSelectedDay = useMemo(() => {
   148→    if (!selectedDate || !scheduleData?.schedules) return null
   149→    const dayName = format(selectedDate, 'EEEE')
   150→    return scheduleData.schedules.find((s) => s.day === dayName) || null
   151→  }, [selectedDate, scheduleData])
   152→
   153→  const timeSlots = useMemo(() => {
   154→    if (!scheduleForSelectedDay) return []
   155→    return scheduleForSelectedDay.timeSlots
   156→  }, [scheduleForSelectedDay])
   157→
   158→  // ── Check if selected date is a holiday ──
   159→  const isHoliday = useMemo(() => {
   160→    if (!selectedDate || !scheduleData?.holidays) return false
   161→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   162→    return scheduleData.holidays.some((h) => h.date === dateStr)
   163→  }, [selectedDate, scheduleData])
   164→
   165→  // When date changes, reset slot selection
   166→  const handleDateSelect = useCallback(
   167→    (date: Date | undefined) => {
   168→      setSelectedDate(date)
   169→      setSelectedSlot(null)
   170→      setShowBookingForm(false)
   171→      if (!date) {
   172→        setSlotStatuses([])
   173→      }
   174→    },
   175→    []
   176→  )
   177→
   178→  // Fetch slot availability when date or slots change (useEffect avoids stale closure)
   179→  useEffect(() => {
   180→    if (!selectedDate || timeSlots.length === 0 || !scheduleData?.doctorId) {
   181→      return
   182→    }
   183→
   184→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   185→    const currentDoctorId = scheduleData.doctorId
   186→    const currentSlots = [...timeSlots]
   187→    let cancelled = false
   188→
   189→    fetch(`/api/patient/bookings/slots-availability?doctorId=${currentDoctorId}&date=${dateStr}`)
   190→      .then((r) => {
   191→        if (!r.ok) {
   192→          // Auth or server error — show all slots as available but log warning
   193→          console.warn('Slots availability API returned', r.status)
   194→          return null
   195→        }
   196→        return r.json()
   197→      })
   198→      .then((data) => {
   199→        if (cancelled) return
   200→        if (!data) {
   201→          // API returned non-OK (e.g. 401) — still show slots so user can see them
   202→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: true, loading: false })))
   203→          return
   204→        }
   205→        if (data.available === false) {
   206→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: false, loading: false })))
   207→        } else {
   208→          const bookedSet = new Set(data.bookedSlots || [])
   209→          setSlotStatuses(
   210→            currentSlots.map((slot) => ({
   211→              slot,
   212→              available: !bookedSet.has(slot),
   213→              loading: false,
   214→            }))
   215→          )
   216→        }
   217→      })
   218→      .catch(() => {
   219→        if (!cancelled) {
   220→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: true, loading: false })))
   221→        }
   222→      })
   223→
   224→    return () => {
   225→      cancelled = true
   226→    }
   227→  }, [selectedDate, timeSlots, scheduleData?.doctorId])
   228→

   231→    if (timeSlots.length === 0) return []
   232→    if (slotStatuses.length !== timeSlots.length) return []
   233→    return slotStatuses
   234→  }, [timeSlots, slotStatuses])
   235→

   238→    mutationFn: (body: Record<string, unknown>) =>
   239→      fetch('/api/patient/bookings', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        credentials: 'include',
   243→        body: JSON.stringify(body),

   253→    },
   254→    onError: (err: Error) => {
   255→      toast.error(err.message || 'Failed to book appointment')
   256→    },
   257→  })
   258→

   263→    setSelectedSlot(slot === selectedSlot ? null : slot)
   264→    setShowBookingForm(slot !== selectedSlot || showBookingForm)
   265→  }
   266→
   267→  const handleBook = () => {
   268→    if (!selectedDate || !disease.trim()) {
   269→      toast.error('Please fill in all required fields')
   270→      return
   271→    }
   272→
   273→    if (!scheduleData?.doctorId) {
   274→      toast.error('Schedule data not loaded. Please wait and try again.')
   275→      return
   276→    }
   277→
   278→    const selectedGender = bookingGender || user?.gender || 'Male'
   279→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   280→    bookMutation.mutate({
   281→      doctorId: scheduleData.doctorId,
   282→      bookingDate: dateStr,
   283→      timeSlot: selectedSlot || '',
   284→      bookingMode,
   285→      gender: selectedGender,
   286→      disease: disease.trim(),
   287→      description: description.trim(),
   288→      state: bookingState.trim(),
   289→      city: bookingCity.trim(),
   290→      ...(hospitalId ? { hospitalId } : {}),
   291→      ...(departmentId ? { departmentId } : {}),
   292→    })
   293→  }
   294→
   295→  // ── Calendar disabled dates ──
   296→  const disablePastDates = (date: Date) => {
   297→    return isBefore(startOfDay(date), startOfDay(new Date()))
   298→  }
   299→
   300→  const disableNonScheduleDays = (date: Date) => {
   301→    if (!scheduleData?.schedules) return false // Allow all if no schedule loaded yet
   302→    const dayName = format(date, 'EEEE')
   303→    const hasSchedule = scheduleData.schedules.some((s) => s.day === dayName)
   304→    return !hasSchedule
   305→  }
   306→
   307→  // ── Loading state ──
   308→  if (doctorLoading || scheduleLoading) {
   309→    return <BookingPageSkeleton />
   310→  }
   311→
   312→  if (!doctor) {
   313→    return (
   314→      <div className="flex flex-col items-center justify-center py-20 text-center">
   315→        <AlertCircle className="h-12 w-12 text-muted-foreground/40 mb-4" />
   316→        <h2 className="text-xl font-bold mb-2">Doctor Not Found</h2>
   317→        <p className="text-sm text-muted-foreground mb-6">The doctor you are looking for does not exist.</p>
   318→        <Button variant="outline" className="text-teal-600 border-teal-200 hover:bg-teal-50 dark:border-teal-800 dark:text-teal-400" asChild>
   319→          <Link href="/doctors">
   320→            <ChevronLeft className="h-4 w-4 mr-1" />
   321→            Browse Doctors
   322→          </Link>
   323→        </Button>
   324→      </div>
   325→    )
   326→  }
   327→
   328→  const doc = doctor.doctor
   329→  const initials = doctor.name
   330→    .split(' ')
   331→    .map((n) => n[0])
   332→    .join('')
   333→    .slice(0, 2)
   334→    .toUpperCase()
   335→
   336→  // Doctor has published no availability — every calendar date would be
   337→  // disabled. Show a clear empty state instead of a dead calendar.
   338→  if (!scheduleData?.schedules || scheduleData.schedules.length === 0) {
   339→    return (
   340→      <div className="max-w-4xl mx-auto space-y-6">
   341→        <motion.div {...fadeIn} className="flex items-center gap-2 text-sm text-muted-foreground">
   342→          <Link href="/dashboard/patient" className="hover:text-teal-600 transition-colors">Dashboard</Link>
   343→          <span>/</span>
   344→          <Link href="/dashboard/patient/appointments" className="hover:text-teal-600 transition-colors">Appointments</Link>
   345→          <span>/</span>
   346→          <span className="text-foreground font-medium truncate">Book Appointment</span>
   347→        </motion.div>
   348→
   349→        <motion.div {...fadeIn} transition={{ delay: 0.05 }}>
   350→          <Card>
   351→            <CardContent className="flex flex-col items-center justify-center py-16 text-center">
   352→              <div className="flex h-16 w-16 items-center justify-center rounded-full bg-amber-50 dark:bg-amber-950/30 mb-4">
   353→                <CalendarOff className="h-8 w-8 text-amber-500" />
   354→              </div>
   355→              <h2 className="text-xl font-bold mb-2">No availability published yet</h2>
   356→              <p className="text-sm text-muted-foreground max-w-md mb-6">
   357→                {doctor.name} hasn&apos;t published any consultation schedule yet. Please check
   358→                back later or browse other doctors with open slots.
   359→              </p>
   360→              <div className="flex flex-col sm:flex-row gap-3">
   361→                <Button asChild className="bg-teal-600 hover:bg-teal-700 text-white">
   362→                  <Link href="/doctors">
   363→                    <CalendarDays className="h-4 w-4 mr-2" />
   364→                    Browse Other Doctors
   365→                  </Link>
   366→                </Button>
   367→                <Button variant="outline" asChild>
   368→                  <Link href={`/doctors/${doctorId}`}>
   369→                    View Doctor Profile
   370→                  </Link>
   371→                </Button>
   372→              </div>
   373→            </CardContent>
   374→          </Card>
   375→        </motion.div>
   376→      </div>
   377→    )
   378→  }
   379→
   380→  return (
   381→    <div className="max-w-4xl mx-auto space-y-6">
   382→      {/* Breadcrumb */}
   383→      <motion.div {...fadeIn} className="flex items-center gap-2 text-sm text-muted-foreground">
   384→        <Link href="/dashboard/patient" className="hover:text-teal-600 transition-colors">Dashboard</Link>
   385→        <span>/</span>
   386→        <Link href="/dashboard/patient/appointments" className="hover:text-teal-600 transition-colors">Appointments</Link>
   387→        <span>/</span>
   388→        <span className="text-foreground font-medium truncate">Book Appointment</span>
   389→      </motion.div>
   390→
   391→      {/* Doctor Info Card */}
   392→      <motion.div {...fadeIn} transition={{ delay: 0.05 }}>
   393→        <Card className="border-teal-200/50 dark:border-teal-800/30 overflow-hidden">
   394→          <div className="bg-gradient-to-r from-teal-500 to-emerald-500 h-2" />
   395→          <CardContent className="p-5 md:p-6">
   396→            <div className="flex items-center gap-4">
   397→              <Avatar className="h-16 w-16 border-2 border-teal-200 dark:border-teal-800 shrink-0">
   398→                <AvatarImage src={getAvatarDisplayUrl(doctor.profileImg)} />
   399→                <AvatarFallback className="bg-gradient-to-br from-teal-100 to-teal-200 text-teal-700 font-bold text-xl">
   400→                  {initials}
   401→                </AvatarFallback>
   402→              </Avatar>
   403→              <div className="flex-1 min-w-0">
   404→                <div className="flex items-center gap-2 flex-wrap">
   405→                  <h2 className="text-xl font-bold">{doctor.name}</h2>
   406→                  <OnlineDoctorDot doctorUserId={doctor.id} doctorName={doctor.name} className="ml-1" />

   409→                    <Badge variant="destructive" className="text-[10px] px-1.5">Emergency</Badge>

   411→                </div>
   412→                <p className="text-teal-600 dark:text-teal-400 font-medium text-sm mt-0.5">
   413→                  {doc?.specialization || 'General Physician'}
   414→                </p>
   415→                <div className="flex items-center gap-4 mt-1.5 text-xs text-muted-foreground flex-wrap">

   419→                    <IndianRupee className="h-3 w-3" />{fees} consultation

   421→                </div>
   422→              </div>
   423→              <Button variant="outline" size="icon" className="shrink-0 h-8 w-8" asChild>
   424→                <Link href={`/doctors/${doctorId}`}>
   425→                  <ChevronLeft className="h-4 w-4" />

   427→                </Link>

   429→            </div>
   430→          </CardContent>

   434→      {/* Main booking area */}

   438→          {/* Step 1: Select Date */}

   441→              <CardHeader className="pb-3">

   444→                  <span className="flex items-center gap-2">
   445→                    Step 1: Select Date

   448→                </CardTitle>
   449→              </CardHeader>

   452→                  <Calendar
   453→                    mode="single"
   454→                    selected={selectedDate}
   455→                    onSelect={handleDateSelect}
   456→                    disabled={(date) => disablePastDates(date) || disableNonScheduleDays(date)}
   457→                    className="rounded-xl border p-3 pointer-events-auto"
   458→                    classNames={{
   459→                      day_selected: 'bg-teal-500 text-white hover:bg-teal-600 hover:text-white focus:bg-teal-500 focus:text-white',
   460→                      day_today: 'bg-teal-100 text-teal-700 dark:bg-teal-900/50 dark:text-teal-300',
   461→                      day_disabled: 'text-muted-foreground/30 line-through',
   462→                    }}
   463→                  />
   464→                </div>
   465→
   466→                {/* Holiday notice */}
   467→                <AnimatePresence>
   468→                  {selectedDate && isHoliday && (
   469→                    <motion.div
   470→                      initial={{ opacity: 0, height: 0 }}
   471→                      animate={{ opacity: 1, height: 'auto' }}
   472→                      exit={{ opacity: 0, height: 0 }}
   473→                      className="mt-3"
   474→                    >
   475→                      <div className="flex items-center gap-2 rounded-lg bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 px-3 py-2 text-sm text-amber-700 dark:text-amber-400">

   478→                      </div>

   480→                  )}
   481→                </AnimatePresence>
   482→              </CardContent>
   483→            </Card>
   484→          </motion.div>
   485→
   486→          {/* Step 2: Select Slot + Mode */}
   487→          <motion.div {...fadeIn} transition={{ delay: 0.15 }}>
   488→            <Card>
   489→              <CardHeader className="pb-3">
   490→                <CardTitle className="text-base font-semibold flex items-center gap-2">
   491→                  <Clock className="h-5 w-5 text-teal-500" />
   492→                  <span className="flex items-center gap-2">

   519→                      <Button
