   127→// Detect whether a string is an image URL/URI (vs plain text header content).
   128→// Accepts http(s) URLs, root-relative paths (/foo.png), and data: image URLs.
   129→function isImageUrl(s: string): boolean {
   130→  if (!s) return false
   131→  const t = s.trim()
   132→  return /^(https?:\/\/|\/[^/\s]+|data:image\/)/i.test(t)
   133→}
   134→
   135→// Split a header text string into display lines (preserving explicit \n).
   136→function splitHeaderLines(s: string): string[] {
   137→  return (s || '').split(/\r?\n/).map((l) => l.trim()).filter(Boolean)
   138→}

   140→export function PrescriptionPrintView({ data, onClose, onPrint, autoPrint = true, extraActions }: PrescriptionPrintViewProps) {

   145→  // named labels fill empty vital slots (pulse/SpO2 are only ever labels);

   152→  // Advice grouped complaint-wise (mirrors wizard Step 5/6 sections).

   154→  const adviceGroups = (() => {
   155→    const coNameMap = new Map<string, { coDetail: string; coDetailEn: string }>()
   156→    for (const c of complaints || []) {
   157→      if (c.coId && (c.coDetail || c.coDetailEn)) {
   158→        coNameMap.set(c.coId, { coDetail: c.coDetail, coDetailEn: c.coDetailEn })
   159→      }

   161→    const all = suggestions || []
   162→    const general = all.filter((s) => !s.coId)
   163→    const other = all.filter((s) => s.coId && !coNameMap.has(s.coId))

   166→      const items = all.filter((s) => s.coId === coId)
   167→      if (items.length > 0) {
   168→        sections.push({ key: coId, name: names.coDetailEn || names.coDetail, items })
   169→      }
   170→    }
   171→    if (other.length > 0) {
   172→      sections.push({ key: '__other__', name: 'Other Advice', items: other })
   173→    }
   174→    if (general.length > 0) {
   175→      sections.push({ key: '__general__', name: 'General Advice', items: general })
   176→    }
   177→    return sections
   178→  })()
   179→
   180→  useEffect(() => {
   181→    if (!autoPrint) return
   182→    const timer = setTimeout(() => {

   184→    }, 150)

   186→  }, [autoPrint, onPrint])
   187→
   188→  // Determine which time columns are needed based on medicine data
   189→  const hasMorning = medicines.some((m) => m.morning > 0)
   190→  const hasAfternoon = medicines.some((m) => m.afternoon > 0)
   191→  const hasEvening = medicines.some((m) => m.evening > 0)

   198→          body * { visibility: hidden; }
   199→          .prescription-print-area, .prescription-print-area * { visibility: visible; }
   200→          .prescription-print-area { position: absolute; left: 0; top: 0; width: 100%; }
   201→          .print-close-btn, .print-actions-bar { display: none !important; }
   202→          @page { size: A4; margin: 10mm; }
   203→        }
   204→      ` }} />
   205→
   206→      <AnimatePresence>
   207→        <motion.div
   208→          className="fixed inset-0 z-50 flex items-start justify-center bg-black/60 overflow-y-auto py-8"
   209→          initial={{ opacity: 0 }}
   210→          animate={{ opacity: 1 }}
   211→          exit={{ opacity: 0 }}
   212→          onClick={(e) => { if (e.target === e.currentTarget) onClose() }}
   213→        >
   214→          {/* Extra actions bar (e.g. Print + Mark Visit Complete) - hidden in print */}
   215→          {extraActions && (
   216→            <div className="print-actions-bar fixed top-4 left-1/2 z-[60] -translate-x-1/2 flex items-center gap-2 rounded-full bg-white/90 px-3 py-1.5 shadow-lg">
   217→              {extraActions}
   218→            </div>
   219→          )}
   220→
   221→          {/* Close button - hidden in print */}

   223→            onClick={onClose}
   224→            className="print-close-btn fixed top-4 right-4 z-[60] rounded-full bg-white/90 p-2 shadow-lg hover:bg-white transition-colors"
   225→            aria-label="Close print preview"
   226→          >
   227→            <X className="h-5 w-5 text-gray-600" />
   228→          </button>
   229→
   230→          {/* Prescription document */}
   231→          <motion.div
   232→            className="prescription-print-area bg-white rounded-lg shadow-2xl"
   233→            style={{ width: '210mm', maxWidth: '95vw' }}

   241→              {/* ── Header ── */}

   244→                {settings.isFullHeader && settings.fullHeader ? (

   246→                    /* Image letterhead (e.g. uploaded clinic letterhead PNG) */

   251→                        style={{ maxHeight: '120px', maxWidth: '100%', objectFit: 'contain' }}

   267→                    <div style={{ textAlign: 'center' }}>
   268→                      {splitHeaderLines(settings.fullHeader).map((line, i, arr) => (

   282→                      {/* Hospital context line below text header */}
   283→                      {hospital && hospital.hospitalName && (

   285→                          <p style={{ margin: 0, fontSize: '14px', fontWeight: 700, color: COLORS.text }}>{hospital.hospitalName}</p>
   286→                          {(hospital.departmentName || hospital.departmentFloor || hospital.departmentOpdRoom) && (
   287→                            <p style={{ margin: '2px 0 0 0', fontSize: '12px', color: COLORS.gray500, fontWeight: 500 }}>
   288→                              {[hospital.departmentName, hospital.departmentFloor && `Floor ${hospital.departmentFloor}`, hospital.departmentOpdRoom].filter(Boolean).join(' · ')}
   289→                            </p>
   290→                          )}
   291→                        </div>
   292→                      )}
   293→                    </div>
   294→                  )
   295→                ) : (
   296→                  /* Standard header: logo + doctor info text */
   297→                  <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
   298→                    <div style={{ display: 'flex', gap: '12px', alignItems: 'flex-start' }}>
   299→                      {settings.logo && (
   300→                        <div style={{ flexShrink: 0 }}>
   301→                          <img
   302→                            src={settings.logo}
   303→                            alt="Logo"
   304→                            style={{ height: '60px', width: '60px', objectFit: 'contain', borderRadius: '4px' }}
   305→                          />
   306→                        </div>
   307→                      )}
   308→                      <div>
   309→                        {/* Hospital context — shown above doctor name when available */}
   310→                        {hospital && hospital.hospitalName && (
   311→                          <div style={{ marginBottom: '6px' }}>
   312→                            <h2 style={{ margin: '0 0 3px 0', fontSize: '18px', fontWeight: 800, color: COLORS.text, letterSpacing: '0.2px' }}>
   313→                              {hospital.hospitalName}
   314→                            </h2>
   315→                            {(hospital.departmentName || hospital.departmentFloor || hospital.departmentOpdRoom) && (
   316→                              <p style={{ margin: 0, fontSize: '12px', color: COLORS.gray500, fontWeight: 500 }}>
   317→                                {[hospital.departmentName, hospital.departmentFloor && `Floor ${hospital.departmentFloor}`, hospital.departmentOpdRoom].filter(Boolean).join(' · ')}
   318→                              </p>
   319→                            )}
   320→                          </div>
   321→                        )}
   322→
   323→                        {/* Doctor-supplied custom header text takes precedence over auto-generated block */}
   324→                        {settings.header && settings.header.trim() ? (
   325→                          <div>
   326→                            {splitHeaderLines(settings.header).map((line, i, arr) => (
   327→                              <p
   328→                                key={i}
   329→                                style={{
   330→                                  margin: i === 0 ? '0 0 3px 0' : i === arr.length - 1 ? '0' : '0 0 3px 0',
   331→                                  fontSize: i === 0 ? '20px' : '12px',
   332→                                  fontWeight: i === 0 ? 700 : 500,
   333→                                  color: i === 0 ? COLORS.teal : COLORS.gray500,
   334→                                  letterSpacing: i === 0 ? '0.3px' : 'normal',
   335→                                }}
   336→                              >
   337→                                {line}
   338→                              </p>
   339→                            ))}
   340→                          </div>
   341→                        ) : (
   342→                          /* Auto-generated header from doctor profile */
   343→                          <>
   344→                            <h1 style={{ margin: '0 0 2px 0', fontSize: '20px', fontWeight: 700, color: COLORS.teal, letterSpacing: '0.3px' }}>
   345→                              {doctorDisplayName(doctor.name) || 'Unknown'}
   346→                            </h1>
   347→                            {doctor.specialization && (
   348→                              <p style={{ margin: '0 0 3px 0', fontSize: '13px', color: COLORS.gray700, fontWeight: 500 }}>
   349→                                {doctor.specialization}
   350→                              </p>
   351→                            )}
   352→                            {doctor.education && (
   353→                              <p style={{ margin: '0 0 3px 0', fontSize: '12px', color: COLORS.gray500 }}>
   354→                                {doctor.education}
   355→                              </p>
   356→                            )}
   357→                            {doctor.registrationDetail && (
   358→                              <p style={{ margin: '0', fontSize: '11px', color: COLORS.gray500 }}>
   359→                                Reg. No: {doctor.registrationDetail}
   360→                              </p>
   361→                            )}
   362→                          </>
   363→                        )}
   364→                      </div>
   365→                    </div>
   366→                    <div style={{ textAlign: 'right' }}>
   367→                      <p style={{ margin: '0 0 4px 0', fontSize: '12px', color: COLORS.gray700, fontWeight: 600 }}>
   368→                        PRESCRIPTION
   369→                      </p>
   370→                      <p style={{ margin: '0 0 2px 0', fontSize: '11px', color: COLORS.gray500 }}>
   371→                        Date: {format(new Date(createdAt), 'dd MMM yyyy')}
   372→                      </p>
   373→                      {(doctor.phoneNo || doctor.mobileNo) && (
   374→                        <p style={{ margin: '0', fontSize: '11px', color: COLORS.gray500 }}>
   375→                          Ph: {doctor.phoneNo || doctor.mobileNo}
   376→                        </p>
   377→                      )}
   378→                    </div>
   379→                  </div>
   380→                )}
   381→                {/* Address line */}
   382→                {(doctor.hospitalAddress || doctor.address) && (
   383→                  <p style={{ margin: settings.isFullHeader && settings.fullHeader ? '4px 0 0 0' : '8px 0 0 0', fontSize: '11px', color: COLORS.gray500 }}>
   384→                    {doctor.hospitalAddress || doctor.address}{doctor.city ? `, ${doctor.city}` : ''}{doctor.state ? `, ${doctor.state}` : ''}
   385→                  </p>
   386→                )}
   387→              </div>
   388→
   389→              {/* ── Patient Info ── */}
   390→              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(140px, 1fr))', gap: '6px 24px', padding: '10px 14px', backgroundColor: COLORS.gray100, borderRadius: '6px', marginBottom: '16px', fontSize: '12px' }}>
   391→                <div>
   392→                  <span style={{ color: COLORS.gray500 }}>Patient: </span>
   393→                  <span style={{ fontWeight: 600 }}>{patient.name}</span>
   394→                </div>
   395→                {patient.age && (
   396→                  <div>
   397→                    <span style={{ color: COLORS.gray500 }}>Age: </span>
   398→                    <span style={{ fontWeight: 500 }}>{patient.age}</span>
   399→                  </div>
   400→                )}
   401→                {patient.gender && (
   402→                  <div>
   403→                    <span style={{ color: COLORS.gray500 }}>Gender: </span>
   404→                    <span style={{ fontWeight: 500 }}>{patient.gender}</span>

   426→              {(mergedVitals.bp || mergedVitals.temperature || mergedVitals.weight || mergedVitals.pulse || mergedVitals.spo2 || extraLabels.length > 0) && (
