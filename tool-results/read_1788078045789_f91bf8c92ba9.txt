   132→// Detect whether a string is an image URL/URI (vs plain text header content).
   133→// Accepts http(s) URLs, root-relative paths (/foo.png), and data: image URLs.
   134→function isImageUrl(s: string): boolean {
   135→  if (!s) return false
   136→  const t = s.trim()
   137→  return /^(https?:\/\/|\/[^/\s]+|data:image\/)/i.test(t)
   138→}
   139→
   140→// Split a header text string into display lines (preserving explicit \n).
   141→function splitHeaderLines(s: string): string[] {
   142→  return (s || '').split(/\r?\n/).map((l) => l.trim()).filter(Boolean)
   143→}

   145→export function PrescriptionPrintView({ data, onClose, onPrint, autoPrint = true, extraActions }: PrescriptionPrintViewProps) {

   150→
   151→  // Advice grouped complaint-wise (mirrors wizard Step 5/6 sections).

   156→      if (c.coId && (c.coDetail || c.coDetailEn)) {
   157→        coNameMap.set(c.coId, { coDetail: c.coDetail, coDetailEn: c.coDetailEn })
   158→      }
   159→    }
   160→    const all = suggestions || []
   161→    const general = all.filter((s) => !s.coId)
   162→    const other = all.filter((s) => s.coId && !coNameMap.has(s.coId))

   165→      const items = all.filter((s) => s.coId === coId)
   166→      if (items.length > 0) {
   167→        sections.push({ key: coId, name: names.coDetailEn || names.coDetail, items })
   168→      }
   169→    }
   170→    if (other.length > 0) {
   171→      sections.push({ key: '__other__', name: 'Other Advice', items: other })
   172→    }
   173→    if (general.length > 0) {
   174→      sections.push({ key: '__general__', name: 'General Advice', items: general })
   175→    }
   176→    return sections
   177→  })()

   180→    if (!autoPrint) return
   181→    const timer = setTimeout(() => {
   182→      onPrint()

   184→    return () => clearTimeout(timer)
   185→  }, [autoPrint, onPrint])
   186→
   187→  // Determine which time columns are needed based on medicine data
   188→  const hasMorning = medicines.some((m) => m.morning > 0)

   190→  const hasEvening = medicines.some((m) => m.evening > 0)

   198→          .prescription-print-area, .prescription-print-area * { visibility: visible; }
   199→          .prescription-print-area { position: absolute; left: 0; top: 0; width: 100%; }
   200→          .print-close-btn, .print-actions-bar { display: none !important; }
   201→          @page { size: A4; margin: 10mm; }
   202→        }
   203→      ` }} />
   204→
   205→      <AnimatePresence>
   206→        <motion.div
   207→          className="fixed inset-0 z-50 flex items-start justify-center bg-black/60 overflow-y-auto py-8"
   208→          initial={{ opacity: 0 }}
   209→          animate={{ opacity: 1 }}
   210→          exit={{ opacity: 0 }}
   211→          onClick={(e) => { if (e.target === e.currentTarget) onClose() }}
   212→        >
   213→          {/* Extra actions bar (e.g. Print + Mark Visit Complete) - hidden in print */}
   214→          {extraActions && (
   215→            <div className="print-actions-bar fixed top-4 left-1/2 z-[60] -translate-x-1/2 flex items-center gap-2 rounded-full bg-white/90 px-3 py-1.5 shadow-lg">
   216→              {extraActions}
   217→            </div>
   218→          )}
   219→
   220→          {/* Close button - hidden in print */}
   221→          <button
   222→            onClick={onClose}
   223→            className="print-close-btn fixed top-4 right-4 z-[60] rounded-full bg-white/90 p-2 shadow-lg hover:bg-white transition-colors"
   224→            aria-label="Close print preview"
   225→          >
   226→            <X className="h-5 w-5 text-gray-600" />
   227→          </button>
   228→
   229→          {/* Prescription document */}
   230→          <motion.div

   233→            initial={{ opacity: 0, y: 30, scale: 0.97 }}
   234→            animate={{ opacity: 1, y: 0, scale: 1 }}
   235→            exit={{ opacity: 0, y: 30, scale: 0.97 }}
   236→            transition={{ type: 'spring', stiffness: 300, damping: 30 }}

   238→            <div style={{ padding: '24px 32px', fontFamily: '"Segoe UI", Tahoma, Geneva, Verdana, sans-serif', color: COLORS.text, fontSize: '13px', lineHeight: '1.5' }}>
   239→
   240→              {/* ── Header ── */}
   241→              <div style={{ borderBottom: `2px solid ${COLORS.teal}`, paddingBottom: '14px', marginBottom: '16px' }}>

   243→                {settings.isFullHeader && settings.fullHeader ? (

   250→                        style={{ maxHeight: '120px', maxWidth: '100%', objectFit: 'contain' }}
   251→                      />
   252→                      {/* Hospital context line below full header image */}
   253→                      {hospital && hospital.hospitalName && (

   257→                            <p style={{ margin: '2px 0 0 0', fontSize: '12px', color: COLORS.gray500, fontWeight: 500 }}>

   261→                        </div>
   262→                      )}
   263→                    </div>
   264→                  ) : (

   266→                    <div style={{ textAlign: 'center' }}>
   267→                      {splitHeaderLines(settings.fullHeader).map((line, i, arr) => (
   268→                        <p

   270→                          style={{
   271→                            margin: i === 0 ? '0 0 2px 0' : i === arr.length - 1 ? '2px 0 0 0' : '1px 0',
   272→                            fontSize: i === 0 ? '20px' : '12px',
   273→                            fontWeight: i === 0 ? 800 : 500,
   274→                            color: i === 0 ? COLORS.teal : COLORS.gray700,
   275→                            letterSpacing: i === 0 ? '0.3px' : 'normal',
   276→                          }}
   277→                        >
   278→                          {line}
   279→                        </p>
   280→                      ))}
   281→                      {/* Hospital context line below text header */}
   282→                      {hospital && hospital.hospitalName && (
   283→                        <div style={{ marginTop: '6px' }}>
   284→                          <p style={{ margin: 0, fontSize: '14px', fontWeight: 700, color: COLORS.text }}>{hospital.hospitalName}</p>
   285→                          {(hospital.departmentName || hospital.departmentFloor || hospital.departmentOpdRoom) && (
   286→                            <p style={{ margin: '2px 0 0 0', fontSize: '12px', color: COLORS.gray500, fontWeight: 500 }}>
   287→                              {[hospital.departmentName, hospital.departmentFloor && `Floor ${hospital.departmentFloor}`, hospital.departmentOpdRoom].filter(Boolean).join(' · ')}
   288→                            </p>
   289→                          )}
   290→                        </div>
   291→                      )}
   292→                    </div>
   293→                  )
   294→                ) : (
   295→                  /* Standard header: logo + doctor info text */
   296→                  <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
   297→                    <div style={{ display: 'flex', gap: '12px', alignItems: 'flex-start' }}>
   298→                      {settings.logo && (
   299→                        <div style={{ flexShrink: 0 }}>
   300→                          <img
   301→                            src={settings.logo}
   302→                            alt="Logo"
   303→                            style={{ height: '60px', width: '60px', objectFit: 'contain', borderRadius: '4px' }}
   304→                          />
   305→                        </div>
   306→                      )}
   307→                      <div>
   308→                        {/* Hospital context — shown above doctor name when available */}
   309→                        {hospital && hospital.hospitalName && (
   310→                          <div style={{ marginBottom: '6px' }}>
   311→                            <h2 style={{ margin: '0 0 3px 0', fontSize: '18px', fontWeight: 800, color: COLORS.text, letterSpacing: '0.2px' }}>
   312→                              {hospital.hospitalName}
   313→                            </h2>
   314→                            {(hospital.departmentName || hospital.departmentFloor || hospital.departmentOpdRoom) && (
   315→                              <p style={{ margin: 0, fontSize: '12px', color: COLORS.gray500, fontWeight: 500 }}>
   316→                                {[hospital.departmentName, hospital.departmentFloor && `Floor ${hospital.departmentFloor}`, hospital.departmentOpdRoom].filter(Boolean).join(' · ')}
   317→                              </p>
   318→                            )}
   319→                          </div>
   320→                        )}
   321→
   322→                        {/* Doctor-supplied custom header text takes precedence over auto-generated block */}
   323→                        {settings.header && settings.header.trim() ? (
   324→                          <div>
   325→                            {splitHeaderLines(settings.header).map((line, i, arr) => (
   326→                              <p
   327→                                key={i}
   328→                                style={{
   329→                                  margin: i === 0 ? '0 0 3px 0' : i === arr.length - 1 ? '0' : '0 0 3px 0',
   330→                                  fontSize: i === 0 ? '20px' : '12px',
   331→                                  fontWeight: i === 0 ? 700 : 500,
   332→                                  color: i === 0 ? COLORS.teal : COLORS.gray500,
   333→                                  letterSpacing: i === 0 ? '0.3px' : 'normal',
   334→                                }}
   335→                              >
   336→                                {line}
   337→                              </p>
   338→                            ))}
   339→                          </div>
   340→                        ) : (
   341→                          /* Auto-generated header from doctor profile */
   342→                          <>
   343→                            <h1 style={{ margin: '0 0 2px 0', fontSize: '20px', fontWeight: 700, color: COLORS.teal, letterSpacing: '0.3px' }}>
   344→                              {doctorDisplayName(doctor.name) || 'Unknown'}
   345→                            </h1>
   346→                            {doctor.specialization && (
   347→                              <p style={{ margin: '0 0 3px 0', fontSize: '13px', color: COLORS.gray700, fontWeight: 500 }}>
   348→                                {doctor.specialization}
   349→                              </p>
   350→                            )}
   351→                            {doctor.education && (
   352→                              <p style={{ margin: '0 0 3px 0', fontSize: '12px', color: COLORS.gray500 }}>
   353→                                {doctor.education}
   354→                              </p>
   355→                            )}
   356→                            {doctor.registrationDetail && (
   357→                              <p style={{ margin: '0', fontSize: '11px', color: COLORS.gray500 }}>
   358→                                Reg. No: {doctor.registrationDetail}
   359→                              </p>
   360→                            )}
   361→                          </>
   362→                        )}
   363→                      </div>
   364→                    </div>
   365→                    <div style={{ textAlign: 'right' }}>
   366→                      <p style={{ margin: '0 0 4px 0', fontSize: '12px', color: COLORS.gray700, fontWeight: 600 }}>
   367→                        PRESCRIPTION
   368→                      </p>
   369→                      <p style={{ margin: '0 0 2px 0', fontSize: '11px', color: COLORS.gray500 }}>
   370→                        Date: {format(new Date(createdAt), 'dd MMM yyyy')}
   371→                      </p>
   372→                      {(doctor.phoneNo || doctor.mobileNo) && (
   373→                        <p style={{ margin: '0', fontSize: '11px', color: COLORS.gray500 }}>
   374→                          Ph: {doctor.phoneNo || doctor.mobileNo}
   375→                        </p>
   376→                      )}
   377→                    </div>
   378→                  </div>
   379→                )}
   380→                {/* Address line */}
   381→                {(doctor.hospitalAddress || doctor.address) && (
   382→                  <p style={{ margin: settings.isFullHeader && settings.fullHeader ? '4px 0 0 0' : '8px 0 0 0', fontSize: '11px', color: COLORS.gray500 }}>
   383→                    {doctor.hospitalAddress || doctor.address}{doctor.city ? `, ${doctor.city}` : ''}{doctor.state ? `, ${doctor.state}` : ''}
   384→                  </p>
   385→                )}
   386→              </div>
   387→
   388→              {/* ── Patient Info ── */}
   389→              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(140px, 1fr))', gap: '6px 24px', padding: '10px 14px', backgroundColor: COLORS.gray100, borderRadius: '6px', marginBottom: '16px', fontSize: '12px' }}>
   390→                <div>
   391→                  <span style={{ color: COLORS.gray500 }}>Patient: </span>
   392→                  <span style={{ fontWeight: 600 }}>{patient.name}</span>
   393→                </div>
   394→                {patient.age && (
   395→                  <div>
   396→                    <span style={{ color: COLORS.gray500 }}>Age: </span>
   397→                    <span style={{ fontWeight: 500 }}>{patient.age}</span>
   398→                  </div>
   399→                )}
   400→                {patient.gender && (
   401→                  <div>
   402→                    <span style={{ color: COLORS.gray500 }}>Gender: </span>
   403→                    <span style={{ fontWeight: 500 }}>{patient.gender}</span>
   404→                  </div>
   405→                )}
   406→                {patient.bloodGroup && (
   407→                  <div>
   408→                    <span style={{ color: COLORS.gray500 }}>Blood Group: </span>
   409→                    <span style={{ fontWeight: 500 }}>{patient.bloodGroup}</span>
   410→                  </div>
   411→                )}
   412→              </div>
   413→
   414→              {/* ── C/O Section (Chief Complaints) ── */}
   415→              {settings.showCoInPrint && complaints && complaints.length > 0 && (
   416→                <div style={{ marginBottom: '14px' }}>
   417→                  <h3 style={sectionTitle}>Chief Complaints</h3>
   418→                  <p style={{ margin: 0, fontSize: '13px', color: COLORS.text }}>
   419→                    C/O: {complaints.map((c) => c.coDetailEn || c.coDetail).filter(Boolean).join(', ')}

   422→              )}
   423→

   425→              {(vitals.bp || vitals.temperature || vitals.weight || vitals.pulse || vitals.spo2) && (
   426→                <div style={{ marginBottom: '14px' }}>
   427→                  <h3 style={sectionTitle}>Vitals</h3>
   428→                  <div style={{ display: 'flex', gap: '10px', flexWrap: 'wrap' }}>
   429→                    {vitals.weight && (
   430→                      <div style={{ padding: '5px 12px', border: `1px solid ${COLORS.border}`, borderRadius: '4px', fontSize: '12px' }}>
   431→                        <span style={{ color: COLORS.gray500 }}>Weight: </span>

   487→
   488→              {/* ── Medicines Table ── */}
