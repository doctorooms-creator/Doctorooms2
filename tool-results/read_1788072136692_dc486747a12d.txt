   108→

   110→  const queryClient = useQueryClient()
   111→  const queueRef = useRef<HTMLDivElement>(null)
   112→
   113→  // Form state
   114→  const [patientName, setPatientName] = useState('')
   115→  const [mobileNo, setMobileNo] = useState('')
   116→  const [gender, setGender] = useState('')
   117→  const [age, setAge] = useState('')
   118→  const [disease, setDisease] = useState('')
   119→  const [timeSlot, setTimeSlot] = useState('')
   120→  const [bookingMode, setBookingMode] = useState<string>('InPerson')
   121→  const [lastAddedId, setLastAddedId] = useState<string | null>(null)
   122→
   123→  // Hospital mode state
   124→  const [selectedDepartmentId, setSelectedDepartmentId] = useState<string>('')
   125→  const [selectedDoctorId, setSelectedDoctorId] = useState<string>('')
   126→  const [queueDoctorFilter, setQueueDoctorFilter] = useState<string>('all')
   127→
   128→  // Fetch today's queue (returns isHospitalMode flag)
   129→  const { data: queueData, isLoading: queueLoading } = useQuery<QueueData>({
   130→    queryKey: ['walkin-queue'],
   131→    queryFn: () =>
   132→      fetch('/api/dashboard/receptionist/walk-in').then((r) => r.json()),
   133→    refetchInterval: 15_000,
   134→  })
   135→
   136→  const isHospitalMode = queueData?.isHospitalMode === true
   137→
   138→  // Fetch departments & doctors for hospital mode
   139→  const { data: hospitalScheduleData, isLoading: hospitalScheduleLoading } = useQuery<{
   140→    isHospitalMode: boolean
   141→    departments: Department[]
   142→    todayName: string
   143→  }>({
   144→    queryKey: ['walkin-hospital-schedule'],
   145→    queryFn: () =>
   146→      fetch('/api/dashboard/receptionist/schedule').then((r) => r.json()),
   147→    enabled: isHospitalMode,
   148→  })
   149→
   150→  const departments = hospitalScheduleData?.departments ?? []
   151→
   152→  // Get selected department's doctors
   153→  const departmentDoctors = useMemo(() => {
   154→    if (!selectedDepartmentId) return []
   155→    const dept = departments.find(
   156→      (d) => d.department.id === selectedDepartmentId
   157→    )
   158→    return dept?.doctors ?? []
   159→  }, [departments, selectedDepartmentId])
   160→
   161→  // Fetch clinic-mode doctor schedule (gives us the linked doctor's id for
   162→  // the live slot inventory)
   163→  const { data: clinicScheduleResponse, isLoading: clinicScheduleLoading } =
   164→    useQuery<{
   165→      doctor?: { id: string; name: string }
   166→      schedules: (ScheduleSlot | null)[]
   167→      todayName: string
   168→    }>({
   169→      queryKey: ['walkin-doctor-schedule'],
   170→      queryFn: () =>
   171→        fetch('/api/dashboard/receptionist/schedule').then((r) => r.json()),
   172→      enabled: !isHospitalMode,
   173→    })
   174→
   175→  // ── Live slot inventory (CTO Plan Phase 2, item 2c) ──
   176→  // Today's date in IST (YYYY-MM-DD) — the same timezone the slot API and
   177→  // queue endpoints use. Recomputed every render so a page left open past
   178→  // midnight IST rolls over on the next poll.
   179→  const todayStr = new Intl.DateTimeFormat('en-CA', {
   180→    timeZone: 'Asia/Calcutta',
   181→  }).format(new Date())
   182→
   183→  // Resolve the doctor whose slots we show: clinic mode → the linked
   184→  // doctor, hospital mode → the doctor selected in the form.
   185→  const slotDoctorId = isHospitalMode
   186→    ? selectedDoctorId
   187→    : (clinicScheduleResponse?.doctor?.id ?? '')
   188→
   189→  const {
   190→    data: slotInventory,
   191→    isLoading: slotsLoading,
   192→    isError: slotsError,
   193→    refetch: refetchSlots,
   194→  } = useQuery<SlotInventory>({
   195→    queryKey: ['slot-inventory', slotDoctorId, todayStr],
   196→    queryFn: async () => {
   197→      const res = await fetch(
   198→        `/api/slots?doctorId=${encodeURIComponent(slotDoctorId)}&date=${todayStr}`
   199→      )
   200→      if (!res.ok) {
   201→        throw new Error('Failed to load slot inventory')
   202→      }
   203→      return res.json() as Promise<SlotInventory>
   204→    },
   205→    enabled: !!slotDoctorId,
   206→    refetchInterval: 15_000, // matches queue polling
   207→  })
   208→
   209→  // Slot-selection bookkeeping. `timeSlot` stays '' for "queue tail" so the
   210→  // submit body (timeSlot || null) keeps making the backend stamp
   211→  // currentTimeIST(). `queueTailPicked` records an explicit tail choice so
   212→  // auto-select never overrides it; `autoPickedSlot` drives the
   213→  // "Auto: next free slot" hint chip.
   214→  const [queueTailPicked, setQueueTailPicked] = useState(false)
   215→  const [autoPickedSlot, setAutoPickedSlot] = useState(false)
   216→
   217→  const resetSlotSelection = useCallback(() => {
   218→    setTimeSlot('')
   219→    setQueueTailPicked(false)
   220→    setAutoPickedSlot(false)
   221→  }, [])
   222→

   224→    setTimeSlot(time)
   225→    setQueueTailPicked(time === '')
   226→    setAutoPickedSlot(false) // manual choice overrides the auto hint
   227→  }, [])
   228→
   229→  // Auto-select "next free": when the inventory loads and the current
   230→  // selection is empty or no longer valid (taken/past/not in the list —
   231→  // e.g. another receptionist claimed it between polls), fall back to the
   232→  // next free slot. Guards: never overrides an explicit "queue tail"
   233→  // choice, and only fires while the current value is empty/invalid, so a
   234→  // valid user selection is never fought.
   235→  useEffect(() => {
   236→    if (!slotInventory || !slotDoctorId) return
   237→    if (queueTailPicked) return

   239→    if (!next) return

   241→    if (!timeSlot || current?.status !== 'free') {
   242→      setTimeSlot(next)
   243→      setAutoPickedSlot(true)
   244→    }
   245→  }, [slotInventory, slotDoctorId, timeSlot, queueTailPicked])
   246→
   247→  // Walk-in mutation
   248→  const walkInMutation = useMutation({
   249→    mutationFn: (body: Record<string, unknown>) =>
   250→      fetch('/api/dashboard/receptionist/walk-in', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify(body),
   254→      }).then((r) => r.json()),
   255→    onSuccess: (result) => {
   256→      if (result.success) {
   257→        if (isHospitalMode && result.booking?.tokenNumber) {
   258→          toast.success(
   259→            `Token ${result.booking.tokenNumber} assigned! Queue #${result.queuePosition}`
   260→          )
   261→        } else {
   262→          toast.success(
   263→            `Patient added to queue #${result.queuePosition}!`
   264→          )
   265→        }
   266→        setLastAddedId(result.booking.id)
   267→        // Reset form
   268→        setPatientName('')
   269→        setMobileNo('')
   270→        setGender('')
   271→        setAge('')
   272→        setDisease('')
   273→        resetSlotSelection() // also re-arms "next free slot" auto-pick
   274→        setBookingMode('InPerson')
   275→        // Reset hospital mode selections
   276→        setSelectedDepartmentId('')
   277→        setSelectedDoctorId('')
   278→        // Invalidate and refetch queue
   279→        queryClient.invalidateQueries({ queryKey: ['walkin-queue'] })
   280→        // Refresh the live slot grid (the booked slot now shows as taken)
   281→        queryClient.invalidateQueries({ queryKey: ['slot-inventory'] })
   282→        queryClient.invalidateQueries({ queryKey: ['receptionist-stats'] })
   283→        queryClient.invalidateQueries({
   284→          queryKey: ['receptionist-appointments'],
   285→        })
   286→        // Scroll to queue after a tick
   287→        setTimeout(() => {
   288→          queueRef.current?.scrollIntoView({
   289→            behavior: 'smooth',
   290→            block: 'start',
   291→          })
   292→        }, 300)
   293→      } else {
   294→        toast.error(result.error || 'Failed to register patient')
   295→        // Slot conflicts (e.g. "Time slot 10:30 is already booked", 409 from
   296→        // the race-safe claim) — refresh the live grid so the newly taken
   297→        // slot shows as booked
   298→        if (result.error && /slot/i.test(result.error)) {
   299→          queryClient.invalidateQueries({ queryKey: ['slot-inventory'] })
   300→        }
   301→      }
   302→    },
   303→    onError: (err: Error) => {
   304→      toast.error(
   305→        err?.message && /slot/i.test(err.message)
   306→          ? err.message
   307→          : 'Failed to register walk-in patient'
   308→      )
   309→      // Our slot view may be stale after a failed submit — refresh the grid
   310→      queryClient.invalidateQueries({ queryKey: ['slot-inventory'] })
   311→    },
   312→  })
   313→
   314→  const handleSubmit = useCallback(

   322→        toast.error('Please select both department and doctor')
   323→        return
   324→      }
   325→
   326→      const body: Record<string, unknown> = {
   327→        patientName: patientName.trim(),
   328→        mobileNo: mobileNo.trim(),
   329→        gender,
   330→        age: age ? parseInt(age, 10) : null,
   331→        disease: disease.trim(),
   332→        timeSlot: timeSlot || null,
   333→        bookingMode,
   334→      }
   335→
   336→      if (isHospitalMode) {
   337→        body.departmentId = selectedDepartmentId
   338→        body.doctorId = selectedDoctorId
   339→      }
   340→
   341→      walkInMutation.mutate(body)
   342→    },
   343→    [
   344→      patientName,
   345→      mobileNo,
   346→      gender,
   347→      age,
   348→      disease,
   349→      timeSlot,
   350→      bookingMode,
   351→      walkInMutation,
   352→      isHospitalMode,
   353→      selectedDepartmentId,
   354→      selectedDoctorId,
   355→    ]
   356→  )
   357→
   358→  const isSubmitting = walkInMutation.isPending
   359→  const queue = queueData?.queue ?? []
   360→  const opdLimit = queueData?.opdLimit ?? 0
   361→  const totalInQueue = queueData?.totalInQueue ?? 0
   362→  const opdCompletedToday = queueData?.opdCompletedToday ?? 0
   363→
   364→  // Filtered queue for hospital mode
   365→  const filteredQueue = useMemo(() => {
   366→    if (!isHospitalMode || queueDoctorFilter === 'all') return queue
   367→    return queue.filter((item) => item.doctorId === queueDoctorFilter)
   368→  }, [queue, isHospitalMode, queueDoctorFilter])
   369→
   370→  // Unique doctors in queue for filter dropdown
   371→  const queueDoctorOptions = useMemo(() => {
   372→    if (!isHospitalMode) return []
   373→    const seen = new Map<string, string>()
   374→    for (const item of queue) {
   375→      if (item.doctorId && item.doctorName) {
   376→        seen.set(item.doctorId, item.doctorName)
   377→      }
   378→    }
   379→    return Array.from(seen.entries()).map(([id, name]) => ({
   380→      id,
   381→      name,
   382→    }))
   383→  }, [queue, isHospitalMode])
   384→
   385→  // Check if form is valid for submit
   386→  const canSubmit =
   387→    patientName.trim() &&
   388→    disease.trim() &&
   389→    (!isHospitalMode || (selectedDepartmentId && selectedDoctorId))
   390→
   391→  return (
   392→    <div className="space-y-6">
   393→      {/* Header */}
   394→      <div className="flex items-center gap-3">
   395→        <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/40">
   396→          <UserPlus className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   397→        </div>
   398→        <div>
   399→          <h2 className="text-xl font-semibold tracking-tight">
   400→            Walk-in Registration
   401→          </h2>
   402→          <p className="text-sm text-muted-foreground">
   403→            {isHospitalMode
   404→              ? 'Register walk-in patients to a department and doctor'
   405→              : 'Register walk-in patients directly to the queue'}
   406→          </p>
   407→        </div>
   408→        {isHospitalMode && (
   409→          <Badge className="ml-2 bg-amber-100 text-amber-700 hover:bg-amber-100 dark:bg-amber-900/40 dark:text-amber-400">
   410→            <Building2 className="mr-1 h-3 w-3" />
   411→            Hospital Mode
   412→          </Badge>
   413→        )}
   414→      </div>
   415→
   416→      <div className="grid gap-6 lg:grid-cols-[380px_1fr]">
   417→        {/* ============ LEFT: Registration Form ============ */}
   418→        <motion.div
   419→          initial={{ opacity: 0, x: -20 }}
   420→          animate={{ opacity: 1, x: 0 }}
   421→          transition={{ duration: 0.3 }}
   422→        >
   423→          <Card>
   424→            <CardHeader className="pb-3">
   425→              <CardTitle className="text-base">Quick Registration</CardTitle>
   426→            </CardHeader>

   431→                  <motion.div

   441→                        className="text-xs font-medium"
   442→                      >
   443→                        <Building2 className="mr-1 inline h-3 w-3" />
   444→                        Department <span className="text-red-500">*</span>
   445→                      </Label>
   446→                      <Select
   447→                        value={selectedDepartmentId}
   448→                        onValueChange={(val) => {
   449→                          setSelectedDepartmentId(val)
   450→                          setSelectedDoctorId('')
   451→                          resetSlotSelection()
   452→                        }}
   453→                        disabled={
   454→                          isSubmitting || hospitalScheduleLoading
   455→                        }
   456→                      >
   457→                        <SelectTrigger className="h-9 w-full">
   458→                          <SelectValue placeholder="Select department" />
   459→                        </SelectTrigger>
   460→                        <SelectContent>
   461→                          {departments.length === 0 &&
   462→                            !hospitalScheduleLoading && (
   463→                              <div className="px-2 py-1.5 text-xs text-muted-foreground">
   464→                                No departments available
   465→                              </div>
   466→                            )}
   467→                          {hospitalScheduleLoading && (
   468→                            <div className="flex items-center gap-2 px-2 py-1.5">

   470→                              <span className="text-xs text-muted-foreground">
   471→                                Loading departments...
   472→                              </span>
   473→                            </div>
   474→                          )}
   475→                          {departments.map((dept) => (
   476→                            <SelectItem
   477→                              key={dept.department.id}
   478→                              value={dept.department.id}

   480→                              <span className="flex items-center gap-2">
   481→                                {dept.department.icon && (

   483→                                )}
   484→                                <span className="font-medium">
   485→                                  {dept.department.name}
   486→                                </span>
   487→                                {dept.department.shortCode && (
   488→                                  <span className="text-muted-foreground">
   489→                                    ({dept.department.shortCode})
   490→                                  </span>
   491→                                )}
   492→                              </span>
   493→                            </SelectItem>
   494→                          ))}
   495→                        </SelectContent>
   496→                      </Select>
   497→                    </div>
   498→
   499→                    {/* Doctor Select */}
   500→                    <div className="space-y-1">
   501→                      <Label
   502→                        htmlFor="doctor"
   503→                        className="text-xs font-medium"
   504→                      >
   505→                        <Stethoscope className="mr-1 inline h-3 w-3" />
   506→                        Doctor <span className="text-red-500">*</span>
   507→                      </Label>
   508→                      <Select
   509→                        value={selectedDoctorId}
   510→                        onValueChange={(val) => {
   511→                          setSelectedDoctorId(val)
   512→                          resetSlotSelection()
   513→                        }}
   514→                        disabled={
   515→                          isSubmitting || !selectedDepartmentId
   516→                        }
   517→                      >
   518→                        <SelectTrigger className="h-9 w-full">
   519→                          <SelectValue placeholder={
   520→                            selectedDepartmentId
   521→                              ? 'Select doctor'
   522→                              : 'Select a department first'
   523→                          } />
   524→                        </SelectTrigger>
   525→                        <SelectContent>
   526→                          {departmentDoctors.length === 0 &&
   527→                            selectedDepartmentId && (
   528→                              <div className="px-2 py-1.5 text-xs text-muted-foreground">
   529→                                No doctors in this department
   530→                              </div>
   531→                            )}
   532→                          {departmentDoctors.map((doc) => (
   533→                            <SelectItem key={doc.id} value={doc.id}>
   534→                              <span className="flex items-center gap-2">
   535→                                <span className="font-medium">
   536→                                  {doc.name}
   537→                                </span>
   538→                                {doc.specialization && (
   539→                                  <span className="text-xs text-muted-foreground">
   540→                                    {doc.specialization}
   541→                                  </span>
   542→                                )}
   543→                              </span>
   544→                            </SelectItem>
   545→                          ))}
   546→                        </SelectContent>
   547→                      </Select>
   548→                    </div>
   549→
   550→                    {/* Divider */}
   551→                    <div className="flex items-center gap-2 pt-1">
   552→                      <div className="h-px flex-1 bg-border" />

   557→                    </div>
   558→                  </motion.div>
   559→                )}
   560→
   561→                {/* Patient Name */}
   562→                <div className="space-y-1">

   564→                    Patient Name <span className="text-red-500">*</span>
   565→                  </Label>
   566→                  <Input
   567→                    id="patientName"
   568→                    placeholder="Full name"
   569→                    value={patientName}
   570→                    onChange={(e) => setPatientName(e.target.value)}
   571→                    disabled={isSubmitting}
   572→                    autoFocus
   573→                    className="h-9"
   574→                  />
   575→                </div>
   576→
   577→                {/* Mobile + Gender row */}
   578→                <div className="grid grid-cols-2 gap-3">

   583→                    <Input
   584→                      id="mobileNo"
   585→                      placeholder="Phone number"
   586→                      value={mobileNo}
