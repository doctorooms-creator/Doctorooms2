   665→   665→
   666→   666→async function createLabBillings(
   667→   667→  createdOrders: Array<{
   668→   668→    id: string;
   669→   669→    orderNo: string;
   670→   670→    spec: OrderSpec;
   671→   671→    labPartnerId: string;
   672→   672→    commissionPercent: number;
   673→   673→    completedAt: Date | null;
   674→   674→  }>,
   675→   675→  drSharma: { id: string },
   676→   676→) {
   677→   677→  console.log('\n💸 Creating Lab Billings (one per Completed order)...');
   678→   678→
   679→   679→  const completed = createdOrders.filter((o) => o.spec.status === 'Completed');
   680→   680→
   681→   681→  // Pick 2 of the 5 to mark Paid (the oldest two: Vitamin B12 #9, Urine Routine #10).
   682→   682→  // Use a Set of order-spec.testName for clarity.
   683→   683→  const paidTestNames = new Set(['Vitamin B12', 'Urine Routine']);
   684→   684→
   685→   685→  let billCounter = 0;
   686→   686→  const createdBillings: Array<{ id: string; orderId: string; paid: boolean }> = [];
   687→   687→
   688→   688→  for (const ord of completed) {
   689→   689→    billCounter++;
   690→   690→    const amount = ord.spec.testFee;
   691→   691→    const commissionAmount = Math.round((amount * ord.commissionPercent) / 100);
   692→   692→    const billedAt = ord.completedAt ?? new Date();
   693→   693→    const isPaid = paidTestNames.has(ord.spec.testName);
   694→   694→    const paidAt = isPaid
   695→   695→      ? new Date(billedAt.getTime() + 3 * 24 * 60 * 60 * 1000) // billedAt + 3 days
   696→   696→      : null;
   697→   697→    const transactionRef = isPaid
   698→   698→      ? `BANK-TRF-2025-${String(billCounter).padStart(3, '0')}`
   699→   699→      : '';
   700→   700→
   701→   701→    const billing = await db.labBilling.create({
   702→   702→      data: {
   703→   703→        labPartnerId: ord.labPartnerId,
   704→   704→        doctorId: drSharma.id,
   705→   705→        patientId: 'dev-patient',
   706→   706→        testOrderId: ord.id,
   707→   707→        amount,
   708→   708→        commissionAmount,
   709→   709→        commissionPercent: ord.commissionPercent,
   710→   710→        paymentStatus: isPaid ? 'Paid' : 'Pending',
   711→   711→        billedAt,
   712→   712→        paidAt,
   713→   713→        transactionRef,
   714→   714→        notes: 'Auto-generated on report upload',
   715→   715→      },
   716→   716→    });
   717→   717→    console.log(
   718→   718→      `  ✓ Billing for ${ord.spec.testName}: ₹${amount} (commission ₹${commissionAmount} @ ${ord.commissionPercent}%) — ${isPaid ? 'Paid' : 'Pending'}` +
   719→   719→      (isPaid ? ` ref=${transactionRef}` : '') +
   720→   720→      ` (id=${billing.id})`
   721→   721→    );
   722→   722→    createdBillings.push({ id: billing.id, orderId: ord.id, paid: isPaid });
   723→   723→  }
   724→   724→
   725→   725→  return { createdBillings };
   726→   726→}
