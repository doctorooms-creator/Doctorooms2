   592→async function createReportUploads(createdOrders: Array<{
   593→  id: string;
   594→  orderNo: string;
   595→  spec: OrderSpec;
   596→  labPartnerId: string;
   597→  completedAt: Date | null;
   598→}>) {
   599→  console.log('\n📄 Creating Lab Report Uploads (one per Completed order)...');
   600→
   601→  const completed = createdOrders.filter(
   602→    (o, idx) => o.spec.status === 'Completed' && REPORT_SPECS[idx + 1] != null,
   603→  );
   604→
   605→  // Map labPartnerId → user (uploadedBy) — the lab partner's own User.id
   606→  const labUserByPartnerId: Record<string, string> = {};
   607→  const partners = await db.labPartner.findMany({ select: { id: true, userId: true } });
   608→  for (const p of partners) labUserByPartnerId[p.id] = p.userId;
   609→
   610→  const createdUploads: Array<{
   611→    id: string;
   612→    orderId: string;
   613→    labPartnerId: string;
   614→    uploadedBy: string;
   615→    completedAt: Date;
   616→    spec: ReportSpec;
   617→  }> = [];
   618→
   619→  for (const ord of completed) {
   620→    const idx = createdOrders.indexOf(ord) + 1;
   621→    const spec = REPORT_SPECS[idx]!;
   622→    const uploadedBy = labUserByPartnerId[ord.labPartnerId] ?? 'dev-lab-tech';
   623→    const completedAt = ord.completedAt ?? new Date();
   624→
   625→    const upload = await db.labReportUpload.create({
   626→      data: {
   627→        externalTestOrderId: ord.id,
   628→        labPartnerId: ord.labPartnerId,
   629→        fileUrl: spec.fileUrl,
   630→        fileName: spec.fileName,
   631→        fileType: spec.fileType,
   632→        fileSize: spec.fileSize,
   633→        reportData: spec.reportData,
   634→        uploadedAt: completedAt,
   635→        uploadedBy,
   636→        verifiedByDoctor: spec.verifiedByDoctor,
   637→        verifiedAt: spec.verifiedByDoctor
   638→          ? new Date(completedAt.getTime() + 24 * 60 * 60 * 1000) // verified 1 day after upload
   639→          : null,
   640→        notes: spec.notes,
   641→      },
   642→    });
   643→    const abnormalTag = spec.notes.startsWith('⚠️') ? ' [ABNORMAL]' : '';
   644→    console.log(
   645→      `  ✓ Report for Order #${idx} (${ord.spec.testName}): ${spec.fileName}` +
   646→      ` (${spec.fileType}, ${Math.round(spec.fileSize / 1000)} KB)` +
   647→      ` verified=${spec.verifiedByDoctor}${abnormalTag} (id=${upload.id})`
   648→    );
   649→    createdUploads.push({
   650→      id: upload.id,
   651→      orderId: ord.id,
   652→      labPartnerId: ord.labPartnerId,
   653→      uploadedBy,
   654→      completedAt,
   655→      spec,
   656→    });
   657→  }
   658→
   659→  return { createdUploads, completedOrders: completed };
   660→}

   663→// 7. LAB BILLINGS (5 — one per Completed order)
   664→// ──────────────────────────────────────────────────────────────
   665→
   666→async function createLabBillings(
   667→  createdOrders: Array<{
   668→    id: string;
   669→    orderNo: string;
   670→    spec: OrderSpec;
   671→    labPartnerId: string;
   672→    commissionPercent: number;
   673→    completedAt: Date | null;
   674→  }>,
   675→  drSharma: { id: string },
   676→) {
   677→  console.log('\n💸 Creating Lab Billings (one per Completed order)...');
   678→
   679→  const completed = createdOrders.filter((o) => o.spec.status === 'Completed');
   680→
   681→  // Pick 2 of the 5 to mark Paid (the oldest two: Vitamin B12 #9, Urine Routine #10).
   682→  // Use a Set of order-spec.testName for clarity.
   683→  const paidTestNames = new Set(['Vitamin B12', 'Urine Routine']);
   684→
   685→  let billCounter = 0;
   686→  const createdBillings: Array<{ id: string; orderId: string; paid: boolean }> = [];
   687→
   688→  for (const ord of completed) {
   689→    billCounter++;
   690→    const amount = ord.spec.testFee;
   691→    const commissionAmount = Math.round((amount * ord.commissionPercent) / 100);
   692→    const billedAt = ord.completedAt ?? new Date();
   693→    const isPaid = paidTestNames.has(ord.spec.testName);
   694→    const paidAt = isPaid
   695→      ? new Date(billedAt.getTime() + 3 * 24 * 60 * 60 * 1000) // billedAt + 3 days
   696→      : null;
   697→    const transactionRef = isPaid
   698→      ? `BANK-TRF-2025-${String(billCounter).padStart(3, '0')}`
   699→      : '';
   700→
   701→    const billing = await db.labBilling.create({
   702→      data: {
   703→        labPartnerId: ord.labPartnerId,
   704→        doctorId: drSharma.id,
   705→        patientId: 'dev-patient',
   706→        testOrderId: ord.id,
   707→        amount,
   708→        commissionAmount,
   709→        commissionPercent: ord.commissionPercent,
   710→        paymentStatus: isPaid ? 'Paid' : 'Pending',
   711→        billedAt,
   712→        paidAt,
   713→        transactionRef,
   714→        notes: 'Auto-generated on report upload',
   715→      },
   716→    });
   717→    console.log(
   718→      `  ✓ Billing for ${ord.spec.testName}: ₹${amount} (commission ₹${commissionAmount} @ ${ord.commissionPercent}%) — ${isPaid ? 'Paid' : 'Pending'}` +
   719→      (isPaid ? ` ref=${transactionRef}` : '') +
   720→      ` (id=${billing.id})`
   721→    );
   722→    createdBillings.push({ id: billing.id, orderId: ord.id, paid: isPaid });
   723→  }
   724→
   725→  return { createdBillings };
   726→}
