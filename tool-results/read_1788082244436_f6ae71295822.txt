   315→/** Small step indicator dot */
   316→function StepDot({ label, active, done }: { label: string; active: boolean; done: boolean }) {
   317→  return (
   318→    <div className="flex flex-col items-center gap-1">
   319→      <motion.div
   320→        animate={done ? { scale: [1, 1.2, 1] } : active ? { scale: [1, 1.1, 1] } : {}}
   321→        transition={{ repeat: active ? Infinity : 0, duration: 1.5 }}
   322→        className={cn(
   323→          'h-3 w-3 rounded-full border-2 transition-colors',
   324→          done
   325→            ? 'bg-teal-500 border-teal-500'
   326→            : active
   327→              ? 'bg-teal-200 border-teal-500 dark:bg-teal-800'
   328→              : 'bg-transparent border-muted-foreground/30'
   329→        )}
   330→      />
   331→      <span className={cn(
   332→        'text-[10px] font-medium transition-colors',
   333→        done ? 'text-teal-600 dark:text-teal-400' : active ? 'text-foreground' : 'text-muted-foreground'
   334→      )}>
   335→        {label}
   336→      </span>
   337→    </div>
   338→  )
   339→}
